"""Interaction-type geometry for the 4-particle framework.

Goal: encode *types* of interactions (motifs), not averaged weights.

The user's framework treats one "node" (facial/biological anchor) as a
time-multiplexed router that can host multiple motifs across circadian windows.
This module provides a physics-grounded motif catalog that can be mapped onto
those nodes without collapsing everything into a single scalar constant.
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, FrozenSet, List, Tuple


PARTICLES: Tuple[str, ...] = ("nu", "p", "e", "gamma")


@dataclass(frozen=True)
class Motif:
    name: str
    participants: FrozenSet[str]  # subset of PARTICLES
    mediator: str  # e.g., "gamma", "Z", "W", "hadronic", "loop"
    level: str  # "tree" | "loop" | "effective"
    kind: str  # "scatter" | "exchange" | "convert" | "bound" | "annihilate" | "create"


def catalog() -> List[Motif]:
    """Minimal motif catalog consistent with standard model interaction structure.

    Notes:
    - We keep the 4 "main" particles as participants and allow mediators outside the set.
    - Proton includes internal hadronic structure; we model that as mediator="hadronic".
    - nu-gamma has no tree-level vertex in the SM; motifs are loop/effective only.
    """

    m: List[Motif] = []

    # Electromagnetic (tree-level QED vertices exist for charged particles).
    m += [
        Motif("e-gamma Compton", frozenset({"e", "gamma"}), "gamma", "tree", "scatter"),
        Motif("e-e Coulomb (Moller/Bhabha class)", frozenset({"e"}), "gamma", "tree", "scatter"),
        Motif("p-gamma Compton / Thomson-like", frozenset({"p", "gamma"}), "gamma", "tree", "scatter"),
        Motif("p-e Coulomb (Rutherford/Mott class)", frozenset({"p", "e"}), "gamma", "tree", "scatter"),
        Motif("gamma-gamma light-by-light", frozenset({"gamma"}), "loop", "loop", "scatter"),
        Motif("e+e- annihilation to gammas (effective class)", frozenset({"e", "gamma"}), "gamma", "effective", "annihilate"),
        Motif("gamma -> e+e- pair creation (effective class)", frozenset({"e", "gamma"}), "gamma", "effective", "create"),
    ]

    # Weak interactions (CC/NC); we keep neutrino participation explicit.
    m += [
        Motif("nu-e neutral current scatter", frozenset({"nu", "e"}), "Z", "tree", "scatter"),
        Motif("nu-e charged current scatter (flavor-dependent)", frozenset({"nu", "e"}), "W", "tree", "scatter"),
        Motif("nu-p neutral current scatter", frozenset({"nu", "p"}), "Z", "tree", "scatter"),
        Motif("nu-p charged current convert (beta-like)", frozenset({"nu", "p", "e"}), "W", "tree", "convert"),
        Motif("e-p capture (beta-like)", frozenset({"e", "p", "nu"}), "W", "tree", "convert"),
        Motif("nu-nu neutral current scatter", frozenset({"nu"}), "Z", "tree", "scatter"),
    ]

    # Proton internal / hadronic structure influences p-p and p-gamma processes.
    m += [
        Motif("p-p residual nuclear (hadronic effective)", frozenset({"p"}), "hadronic", "effective", "bound"),
        Motif("p-gamma photoproduction (hadronic effective)", frozenset({"p", "gamma"}), "hadronic", "effective", "create"),
    ]

    # nu-gamma: no tree-level vertex; only loop/effective couplings.
    m += [
        Motif("nu-gamma loop scatter (effective magnetic moment class)", frozenset({"nu", "gamma"}), "loop", "loop", "scatter"),
    ]

    return m


CATALOG: List[Motif] = catalog()


def motif_counts_by_pair() -> Dict[Tuple[str, str], int]:
    """Count motifs touching an ordered pair (a,b)."""
    out: Dict[Tuple[str, str], int] = {}
    for a in PARTICLES:
        for b in PARTICLES:
            key = (a, b)
            out[key] = 0
            for motif in CATALOG:
                if a in motif.participants and b in motif.participants:
                    out[key] += 1
    return out


def motif_counts_by_edge() -> Dict[FrozenSet[str], int]:
    """Count motifs per undirected edge (set of participants)."""
    edges: Dict[FrozenSet[str], int] = {}
    for motif in CATALOG:
        # Only count edges involving at least two of the four particles.
        if len(motif.participants) < 2:
            continue
        edges[motif.participants] = edges.get(motif.participants, 0) + 1
    return edges


def weakest_edge_by_type_count() -> Tuple[FrozenSet[str], int]:
    edges = motif_counts_by_edge()
    if not edges:
        return frozenset(), 0
    edge = min(edges.items(), key=lambda kv: kv[1])
    return edge[0], edge[1]
