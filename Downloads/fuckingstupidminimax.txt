def apply_8d_overlay(habitat_raster, profile_8d):
    """
    habitat_raster: 5-sphere output raster
    profile_8d: dict {r, h, d, p, s, gamma, g, nu} ∈ [0,1]
    Returns: 8D-modulated habitat map + color encoding
    """
    # 8 dim weight → 8D ecological dimension
    eco_dim = {
        'r': 'rhythm (temporal productivity)',
        'h': 'habitat complexity',
        'd': 'disturbance regime',
        'p': 'productivity',
        's': 'species richness',
        'gamma': 'gene flow',
        'g': 'growth rate',
        'nu': 'niche specialization',
    }
    modulated = habitat_raster.copy()
    for dim, value in profile_8d.items():
        modulated = modulated * (0.5 + value)  # dim value ∈ [0.5, 1.5]
    
    # HSL color encoding
    hue = profile_8d['r'] * 360
    saturation = profile_8d['s']
    lightness = profile_8d['gamma']
    color = f"hsl({hue}, {saturation*100}%, {lightness*100}%)"
    
    return modulated, color
