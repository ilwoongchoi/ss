def apply_channels(channels: dict[str, str]) -> dict[tuple, float]:
    w = {e: float(v) for e, v in BASE_W.items()}
    for ch, state in channels.items():
        if ch not in CHANNEL_MAP:
            continue
        for edge, deltas in CHANNEL_MAP[ch].items():
            delta = float(deltas.get(state, 0.0))
            w[edge] = max(0.0, w[edge] + delta)
    return w


def laplacian(w: dict[tuple, float]) -> np.ndarray:
    L = np.zeros((6, 6), dtype=float)
    for (a, b), wv in w.items():
        ia, ib = IDX[a], IDX[b]
        L[ia, ia] += wv
        L[ib, ib] += wv
        L[ia, ib] -= wv
        L[ib, ia] -= wv
    return L


def project4(x6: np.ndarray) -> np.ndarray:
    """6-subject → 4-coordinate. Quark/gluon load onto SM/BW via 0.2828."""
    q, g, nu, ph, pr, el = x6
    BM = ph + C * g
    BW = pr + C * g
    SM = nu + C * q
    SW = el + C * q
    return np.array([BM, BW, SM, SW], dtype=float)


def day_target() -> np.ndarray:
    deg = np.array([6.0, 8.0, 8.0, 10.0])
    return OMEGA * deg / np.linalg.norm(deg)


def night_target() -> np.ndarray:
    t = day_target().copy()
    t[1] -= C   # BW down
    t[2] += C   # SM up
    return t


def init_state6_from_target4(t4: np.ndarray) -> np.ndarray:
    """Invert projection4 → approximate 6-subject state."""
    BM, BW, SM, SW = t4
    # gluon/quark start at small neutral value
    g = 0.1
    q = 0.1
    ph = BM - C * g
    pr = BW - C * g
    nu = SM - C * q
    el = SW - C * q
    return np.array([q, g, nu, ph, pr, el], dtype=float)


def run_phase(name: str, channels: dict[str, str], steps: int = 64, dt: float = 0.02) -> dict:
    is_night = (name == "hysteresis")
    t4 = night_target() if is_night else day_target()
    x = init_state6_from_target4(t4)

    w = apply_channels(channels)
    L = laplacian(w)

    # Hooke-like attractor in 6D mapped from 4D target
    x_target = init_state6_from_target4(t4)
    stiffness = 0.15

    for _ in range(steps):
        dx = -L @ x + stiffness * (x_target - x)
        x = x + dt * dx
        x = np.clip(x, 0.0, None)

    x4 = project4(x)

    # Spark = photon ignition energy (BM activation: dopamine + androgen + extraversion)
    # NOT quark-gluon release — that's confinement. Spark is EM ignition.
    spark = w[("photon","proton")] + w[("photon","electron")]
    current_qg = w[("quark","gluon")]

    # Z boson proxy = neutrino-proton + neutrino-electron edge weights
    z_proxy = w[("neutrino","proton")] + w[("neutrino","electron")]

    # Fusion rate: BW² × spark × Z_proxy × SM
    BM, BW, SM, SW = x4
    fusion = float((BW**2) * spark * z_proxy * SM)

    return {
        "phase": name,
        "BM": float(BM), "BW": float(BW), "SM": float(SM), "SW": float(SW),
        "omega4": float(np.linalg.norm(x4)),
        "spark": spark,
        "z_proxy": z_proxy,
        "qg_tension": current_qg,
        "fusion_rate": fusion,
        "is_night": is_night,
    }

def closure_score(results: list[dict]) -> tuple[float, dict]:
    r = {d["phase"]: d for d in results}
    p1, p2, hy = r["phase1"], r["phase2"], r["hysteresis"]
    checks = {
        # Phase 1 should be RELATIVELY more BW than Phase 2 (pulsation, not absolute)
        "BW_ratio_P1_gt_P2": (p1["BW"]/max(p1["SM"],1e-9)) > (p2["BW"]/max(p2["SM"],1e-9)),
        "spark_rises_P2":  p2["spark"] > p1["spark"],
        "SM_dominant_HY":  hy["SM"] > hy["BW"],
        "Z_opens_HY":      hy["z_proxy"] > p1["z_proxy"],
        "fusion_peaks_P2": p2["fusion_rate"] >= p1["fusion_rate"],
    }
    return sum(checks.values()) / len(checks), checks
