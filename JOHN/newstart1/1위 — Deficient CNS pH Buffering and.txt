// ── GENERATE ──────────────────────────────────────────────────────────────
async function generate() {
  const desc = document.getElementById('description').value.trim();
  if (!desc) { alert('Describe the visual effect first.'); return; }

  if (!apiKey && used >= FREE) { openAuth(); return; }

  const btn = document.getElementById('gen-btn');

  btn.textContent = '⟳ Generating...';

  const payload = {
    description: desc,
    style: selectedStyle,
    platform: document.getElementById('platform').value,
    priority: document.getElementById('priority').value,
  };

  try {
    const headers = { 'Content-Type': 'application/json' };
    if (apiKey) headers['X-API-Key'] = apiKey;

    const r = await fetch(`${API}/generate`, { method: 'POST', headers, body: JSON.stringify(payload) });

    if (r.status === 401) { openAuth(); return; }
    if (r.status === 429) { alert('Rate limit reached. Upgrade for unlimited access.'); return; }
    if (!r.ok) throw new Error('server');

    const data = await r.json();
    renderResult(data, payload.platform);
    if (!apiKey) { used++; localStorage.setItem('ugl_used', used); updateCredits(); }

  } catch(e) {
    // Demo mode — no server needed to evaluate the UI
    renderDemo(payload);
    if (!apiKey) { used++; localStorage.setItem('ugl_used', used); updateCredits(); }
  } finally {
    btn.disabled = false;
    btn.textContent = '⚡ Generate Optimal Shader';
  }
}

function buildDemoShader(p) {
  const plat = p.platform === 'webgl2' ? '#version 300 es\nprecision highp float;\nuniform float time;\nuniform vec2 res;\nout vec4 fragColor;' : 'precision highp float;\nuniform float time;\nuniform vec2 res;';
  const outVar = p.platform === 'webgl2' ? 'fragColor' : 'gl_FragColor';
  const styleCode = {
    abstract: `
    vec2 p = (gl_FragCoord.xy - 0.5 * res) / res.y;
    // UniformGL: bifurcation-free flow field
    float t = time * 0.4;
    vec2 q = vec2(
      dot(p, vec2(cos(t), -sin(t))),
      dot(p, vec2(sin(t),  cos(t)))
    );
    float r = length(q);
    float a = atan(q.y, q.x);
    // Topology-resolved radial bands — no divergence
    float n = 0.5 + 0.5 * sin(r * 8.0 - t * 2.0 + a * 3.0);
    vec3 col = 0.5 + 0.5 * cos(vec3(0.0, 2.094, 4.189) + n * 6.28 + t);
    ${outVar} = vec4(col, 1.0);`,
    fluid: `
    vec2 p = gl_FragCoord.xy / res;
    float t = time * 0.3;
    // Fluid topology: divergence-free velocity field
    vec2 v = vec2(sin(p.y * 4.0 + t), cos(p.x * 4.0 + t)) * 0.5;
    p += v * 0.1;
    float n = sin(p.x * 6.0 + t) * cos(p.y * 6.0 - t);
    vec3 col = mix(vec3(0.05, 0.12, 0.4), vec3(0.1, 0.8, 0.9), n * 0.5 + 0.5);
    ${outVar} = vec4(col, 1.0);`,
    terrain: `
    vec2 p = gl_FragCoord.xy / res * 4.0 - 2.0;
    float t = time * 0.1;
    // Terrain: deterministic hash — no branching
    float h = fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5);
    h = mix(h, fract(sin(dot(p + 0.1, vec2(269.5, 183.3))) * 43758.5), 0.5);
    vec3 lo = vec3(0.2, 0.5, 0.1); vec3 hi = vec3(0.9, 0.85, 0.7);
    vec3 col = mix(lo, hi, smoothstep(0.3, 0.7, h + 0.1 * sin(t)));
    ${outVar} = vec4(col, 1.0);`,
    particle: `
    vec2 uv = gl_FragCoord.xy / res;
    float t = time * 0.5;
    float col = 0.0;
    // Particle field: vectorised loop, no warp divergence
    for (int i = 0; i < 8; i++) {
      float fi = float(i);
      vec2 pos = vec2(fract(fi * 0.127 + t * 0.1), fract(fi * 0.311 + t * 0.07));
      float d = length(uv - pos);
      col += 0.003 / (d * d + 0.0001);
    }
    vec3 c = vec3(col * 0.5, col * 0.3, col);
    ${outVar} = vec4(c, 1.0);`,
    volumetric: `
    vec2 uv = (gl_FragCoord.xy - 0.5 * res) / res.y;
    float t = time * 0.2;
    float acc = 0.0;
    // Volumetric: ray-marched without branching using smoothstep
    for (int i = 0; i < 16; i++) {
      float z = float(i) * 0.1 + 0.5;
      vec2 q = uv / z;
      float d = length(q - vec2(sin(t), cos(t)) * 0.3);
      acc += smoothstep(0.2, 0.0, d) * 0.12 / z;
    }
    vec3 col = vec3(acc * 0.6, acc * 0.3, acc);
    ${outVar} = vec4(col, 1.0);`,
    geometric: `
    vec2 p = (gl_FragCoord.xy - 0.5 * res) / res.y;
    float t = time * 0.25;
    // Geometric topology: step/mix replace all conditionals
    float r = length(p); float a = atan(p.y, p.x);
    float sides = 6.0;
    float poly = cos(floor(0.5 + a / (6.28318 / sides)) * (6.28318 / sides) - a) * r;
    float edge = smoothstep(0.28, 0.3, poly);
    float ring = smoothstep(0.01, 0.0, abs(poly - 0.3 + 0.05 * sin(t * 3.0)));
    vec3 col = mix(vec3(0.05), vec3(0.4, 0.2, 1.0), edge);
    col += vec3(0.8, 0.9, 1.0) * ring;

// ── WEBGL PREVIEW ─────────────────────────────────────────────────────────
function runPreview(fragSrc, platform) {
  const canvas = document.getElementById('preview-canvas');
  const placeholder = document.getElementById('preview-placeholder');

  if (glAnim) cancelAnimationFrame(glAnim);

  // Ensure WebGL2 context
  if (!glCtx) glCtx = canvas.getContext('webgl2');
  if (!glCtx) { placeholder.innerHTML = '<p>WebGL2 not supported in this browser</p>'; return; }

  const gl = glCtx;
  canvas.width = canvas.offsetWidth * devicePixelRatio;
  canvas.height = 280 * devicePixelRatio;

  const vs = `#version 300 es
void main() {
  vec2 p[3] = vec2[](vec2(-1,-1),vec2(3,-1),vec2(-1,3));
  gl_Position = vec4(p[gl_VertexID], 0, 1);
}`;

  // Ensure fragment shader has #version and correct output for WebGL2 preview
  let fs = fragSrc;
  if (!fs.includes('#version')) fs = '#version 300 es\nprecision highp float;\nuniform float time;\nuniform vec2 res;\nout vec4 fragColor;\n' + fs.replace('void main()', 'void main()');
  if (!fs.includes('out vec4')) {
    fs = fs.replace('#version 300 es', '#version 300 es\nout vec4 fragColor;');
    fs = fs.replace(/gl_FragColor/g, 'fragColor');
  }

  function mkShader(type, src) {
    const s = gl.createShader(type);
    gl.shaderSource(s, src); gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) {
      console.warn('Shader compile:', gl.getShaderInfoLog(s)); return null;
    }
    return s;
  }

  if (glProg) gl.deleteProgram(glProg);
  const prog = gl.createProgram();
  const vs_ = mkShader(gl.VERTEX_SHADER, vs);
  const fs_ = mkShader(gl.FRAGMENT_SHADER, fs);
  if (!vs_ || !fs_) { placeholder.style.display='flex'; canvas.style.display='none'; return; }

  gl.attachShader(prog, vs_); gl.attachShader(prog, fs_);
  gl.linkProgram(prog);
  if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) {
    placeholder.innerHTML = '<div class="icon">⚠</div><p>Preview unavailable for this shader variant</p>';
    placeholder.style.display='flex'; canvas.style.display='none'; return;
  }
  glProg = prog;
  gl.useProgram(prog);

  const tLoc = gl.getUniformLocation(prog, 'time');
  const rLoc = gl.getUniformLocation(prog, 'res');

  placeholder.style.display = 'none';
  canvas.style.display = 'block';

  function frame(t) {
    gl.viewport(0, 0, canvas.width, canvas.height);
    gl.uniform1f(tLoc, t * 0.001);
    gl.uniform2f(rLoc, canvas.width, canvas.height);
    gl.drawArrays(gl.TRIANGLES, 0, 3);
    glAnim = requestAnimationFrame(frame);
  }
  glAnim = requestAnimationFrame(frame);
}

import os

app = FastAPI()

# 환경변수 또는 직접 설정
ALLOWED_ORIGINS = ["https://your-alumni-site.edu"]  # alumni 사이트 도메인

app.add_middleware(
    CORSMiddleware,
    allow_origins=ALLOWED_ORIGINS,
    allow_credentials=True,
    allow_methods=["POST", "GET"],
    allow_headers=["*"],
)

def verify_key(api_key: str):
    if api_key not in API_KEYS:
        raise HTTPException(status_code=403, detail="Invalid API key")
    return API_KEYS[api_key]

@app.middleware("http")
async def check_referrer(request: Request, call_next):
    referrer = request.headers.get("referer", "")
    allowed = any(origin in referrer for origin in ALLOWED_ORIGINS)
    
    # health 체크는 예외
    if request.url.path == "/health":
        return await call_next(request)
    
    if not allowed and not any(origin in str(request.headers.get("origin", "")) for origin in ALLOWED_ORIGINS):
        raise HTTPException(status_code=403, detail="Access denied")
    
    return await call_next(request)

@app.post("/analyze")
async def analyze(request: Request, data: dict):
    # API Key 확인
    api_key = request.headers.get("X-API-Key", "")
    user = verify_key(api_key)
    
    # 분석 로직...
    return {"result": "analysis complete", "user": user}

@app.get("/health")
async def health():
    return {"status": "ok"}

## 해결책

**1. alumni 사이트에서만 iframe으로 embed**

alumni 포털에 iframe 넣고, 직접 URL 접속은 차단:

```python
# server.py에 추가
@app.middleware("http")
async def block_direct_access(request: Request, call_next):
    referrer = request.headers.get("referer", "")
    # iframe에서 오는 요청만 허용
    if not referrer and request.url.path != "/health":
        return HTMLResponse("<h1>Access Denied</h1>", status_code=403)
    return await call_next(request)
```

**2. 또는 API만 열고 UI는 alumni 사이트에 넣기**

- ngrok는 `/analyze` API만 노출
- HTML/JS는 alumni 사이트에 직접 올림
- alumni 사이트에서 `fetch()`로 ngrok API 호출

## 가장 안전한 방법

- ngrok URL로 직접 접속 → 아무것도 안 보임 (API만 있음)

// Phase 1: Static precision mapping
// Tests floating-point fidelity at topological ridge positions.
// Clean hardware → smooth gradient. Low-precision → banding/noise near ridges.
const FS_P1 = `#version 300 es
precision highp float;
uniform vec2  u_res;
out vec4 fragColor;

const float M  = 1.0661;
const float T  = 0.1746;
const float TE = 1.0100375;
const float PH = 1.6180339887;
const float K  = 0.09375;
const float SP = 2.4239;

// Topological ridge positions: cells 5 and 11 of 128-cell manifold
const float R5  = 0.0390625;
const float R11 = 0.0859375;

float ridgeField(vec2 uv) {
    // High-precision oscillator cascade at ridge singularities.
    // All lanes execute identically — no warp divergence.
    float d5  = abs(uv.x - R5)  * M * 48.0;
    float d11 = abs(uv.x - R11) * M * 48.0;

    float osc = sin(d5  * PH * 80.0 + K * TE) *
                cos(d11 * PH * 64.0 - K * TE);
    float y = (uv.y - 0.5) * M;

    float f = osc;
    for (int i = 0; i < 6; i++) {
        f = sin(f * PH * float(i + 2) + K) *
            cos(f * TE * float(i + 1) + y * SP * 0.1);
    }
    return f * 0.5 + 0.5;
}

void main() {
    vec2 uv = gl_FragCoord.xy / u_res;
    float v = ridgeField(uv);
    float v2 = ridgeField(vec2(1.0 - uv.x, uv.y));
    float v3 = ridgeField(vec2(uv.x, 1.0 - uv.y));

    vec3 col = vec3(
        mix(v,  v2, uv.y) * M - 0.06,
        mix(v2, v3, uv.x) * T + 0.10,
        mix(v,  v3, 0.5)  * (1.0 - T) + 0.15

}`;

// Phase 2: Temporal coherence test
// Tests frame-to-frame consistency. Clean hardware → smooth animation.
// Low temporal coherence → shimmer / flickering at edge boundaries.
const FS_P2 = `#version 300 es
precision highp float;
uniform vec2  u_res;
uniform float u_time;
out vec4 fragColor;

const float M  = 1.0661;
const float T  = 0.1746;
const float TE = 1.0100375;
const float PH = 1.6180339887;
const float K  = 0.09375;
const float SP = 2.4239;

float field(vec2 p, float t) {
    float cs = cos(t * T * SP * 0.008);
    float sn = sin(t * T * SP * 0.008);
    vec2 q = vec2(cs * p.x - sn * p.y, sn * p.x + cs * p.y);
    q *= M;

    float v = 0.0, a = 0.5;
    for (int i = 0; i < 7; i++) {
        float fi = float(i);
        v += a * sin(q.x * PH * (fi + 2.0) + t * T * 0.3) *
                 cos(q.y * PH * (fi + 1.5) - t * TE * 0.2);
        q = vec2(q.y, -q.x) * K * 12.0 + q * PH;
        a *= T * 2.2;
    }
    return v * 0.5 + 0.5;
}

void main() {

}`;

// Maximum ALU load. Reveals GPU thermal throttle, VRAM bandwidth limits,
// and accumulated floating-point error at high iteration depth.
const FS_P3 = `#version 300 es
precision highp float;
uniform vec2  u_res;
uniform float u_time;
out vec4 fragColor;

const float M  = 1.0661;
const float T  = 0.1746;
const float TE = 1.0100375;
const float PH = 1.6180339887;
const float K  = 0.09375;
const float SP = 2.4239;

float manifoldCell(vec2 p, float cell_idx) {
    float fi = cell_idx;
    vec2 center = vec2(
        fract(fi * K * M + 0.31) * 2.0 - 1.0,
        fract(fi * T * PH + 0.17) * 2.0 - 1.0

    return sin(d * PH * 8.0 - u_time * T * (fi * 0.06 + 0.4)) * exp(-d * 1.8);

}`;

function compileShader(type, src) {
    const s = gl.createShader(type);
    gl.shaderSource(s, src);
    gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS))
        console.error('Shader compile error:', gl.getShaderInfoLog(s));
    return s;
}

function makeProgram(fsSrc) {
    const prog = gl.createProgram();
    const vs = compileShader(gl.VERTEX_SHADER, VS);
    const fs = compileShader(gl.FRAGMENT_SHADER, fsSrc);
    gl.attachShader(prog, vs);
    gl.attachShader(prog, fs);
    gl.linkProgram(prog);
    return prog;
}

function initGL() {
    glCanvas = document.getElementById('glCanvas');
    gl = glCanvas.getContext('webgl2', { preserveDrawingBuffer: true });
    if (!gl) { alert('WebGL2 not available. Update your browser.'); return false; }

    programs.p1 = makeProgram(FS_P1);
    programs.p2 = makeProgram(FS_P2);
    programs.p3 = makeProgram(FS_P3);

    const verts = new Float32Array([-1,-1, 1,-1, -1,1, 1,1]);
    vbuf = gl.createBuffer();
    gl.bindBuffer(gl.ARRAY_BUFFER, vbuf);
    gl.bufferData(gl.ARRAY_BUFFER, verts, gl.STATIC_DRAW);

    // Allocate pixel readback buffer (sample 64x64 region around center)
    pixelSample_buf = new Uint8Array(64 * 64 * 4);
    return true;
}

function useProgram(prog) {
    gl.useProgram(prog);
    const loc = gl.getAttribLocation(prog, 'a_pos');
    gl.enableVertexAttribArray(loc);
    gl.bindBuffer(gl.ARRAY_BUFFER, vbuf);
    gl.vertexAttribPointer(loc, 2, gl.FLOAT, false, 0, 0);
    return prog;
}

function renderPhase(prog, t) {
    gl.viewport(0, 0, glCanvas.width, glCanvas.height);
    useProgram(prog);
    const uTime = gl.getUniformLocation(prog, 'u_time');
    const uRes  = gl.getUniformLocation(prog, 'u_res');
    if (uTime) gl.uniform1f(uTime, t);
    if (uRes)  gl.uniform2f(uRes, glCanvas.width, glCanvas.height);
    gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
}

function capturePixels() {
    const x0 = Math.floor(glCanvas.width  / 2) - 32;
    const y0 = Math.floor(glCanvas.height / 2) - 32;
    gl.readPixels(x0, y0, 64, 64, gl.RGBA, gl.UNSIGNED_BYTE, pixelSample_buf);
    return new Uint8Array(pixelSample_buf);
}

function pixelVariance(a, b) {
    let diff = 0;
    for (let i = 0; i < a.length; i++) diff += Math.abs(a[i] - b[i]);
    return diff / a.length / 255;
}
