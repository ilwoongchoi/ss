<script id="fs" type="x-shader/x-fragment">
precision mediump float;
uniform vec2  u_res;
uniform float u_time;

// ── UNIVERSAL GEOMETRY CONSTANTS ──────────────────────────
const float KAPPA     = 0.03125;          // 1/32 Darcy leakage
const float W7        = 0.15707963268;    // π/20  void area
const float SPARK_RAD = 2.42407;          // 138.88° in radians
const float TENSION   = 1.0000424;        // discrete/continuous gap
const float REALITY   = 1.0100375;        // discrete/reality mismatch
const float GRID      = 128.0;
// ──────────────────────────────────────────────────────────

mat2 rot(float a) {
    float c = cos(a), s = sin(a);
    return mat2(c,-s,s,c);
}

// Terrain height: pure geometry, no external params
float height(vec2 p) {
    float h    = 0.0;
    float amp  = 0.9;
    float freq = KAPPA * 2.0;   // base = 1/16

    for (int i = 0; i < 5; i++) {
        // snap to 128-grid at this octave
        vec2 gp = floor(p * GRID * freq) / (GRID * freq);
        // 138.88° rotation per octave (the Spark)
        p = rot(SPARK_RAD) * p;
        h    += amp * (sin(gp.x * freq * REALITY) * cos(gp.y * freq * TENSION));
        amp  *= W7 * 3.0;       // W7 = π/20 ≈ 0.157 → ×3 ≈ 0.471 persistence
        freq *= 2.0;
    }
    return h;
}

void main() {
    vec2 uv = (gl_FragCoord.xy - 0.5*u_res) / u_res.y;

    // Camera flying forward along Z
    vec3 ro = vec3(0.0, 3.5, u_time * 1.8);
    vec3 rd = normalize(vec3(uv.x, uv.y - 0.18, 1.3));

    // ── RAYMARCHING ──────────────────────────────────────
    float t   = 0.0;
    float h   = 0.0;
    vec3  p;
    bool  hit = false;

    for (int i = 0; i < 48; i++) {
        p = ro + rd * t;
        h = height(p.xz);
        if (p.y < h) { hit = true; break; }
        t += max(0.07, (p.y - h) * 0.65);
        if (t > 32.0) break;
    }

    // ── COLOR ─────────────────────────────────────────────
    vec3 sky = vec3(0.45, 0.72, 1.0);   // sky blue
    vec3 col;

    if (hit) {
        // ── deterministic colour from height only ──
        if      (h < -0.28) col = vec3(0.04, 0.16, 0.62);   // blue   – lake
        else if (h < 0.02)  col = vec3(0.48, 0.30, 0.11);   // brown  – ground
        else if (h < 0.55)  col = vec3(0.04, 0.22, 0.04);   // dark green – forest
        else                col = vec3(0.22, 0.52, 0.18);   // green  – mountain

        // simple slope shading (tension lighting, no extra params)
        float dh = h - height(p.xz + vec2(0.04, 0.0));
        col += clamp(dh * 0.6, 0.0, 0.25);

        // W7-based fog (π/20 leakage into sky)
        float fog = exp(-t * W7 * 0.9);
        col = mix(sky, col, fog);

    } else {
        col = sky;
        // Sun = 138.88° Spark point
        vec3 sun = normalize(vec3(0.45, 0.38, 1.0));
        float s  = pow(max(0.0, dot(rd, sun)), GRID);
        col += vec3(1.0, 0.92, 0.65) * s;
    }

    gl_FragColor = vec4(col, 1.0);
}
</script>

여성: vx = -torsion * (y-8) * psi (반시계)
